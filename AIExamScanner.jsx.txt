function toB64(file) {
  return new Promise((res,rej)=>{
    const r=new FileReader();
    r.onload=()=>res(r.result.split(",")[1]);
    r.onerror=()=>rej(new Error("read fail"));
    r.readAsDataURL(file);
  });
}

function scoreColor(s){
  if(s>=90) return "score-s";
  if(s>=80) return "score-a";
  if(s>=70) return "score-b";
  if(s>=60) return "score-c";
  return "score-d";
}

function scoreLetter(s){
  if(s>=90) return "A+";
  if(s>=80) return "A";
  if(s>=70) return "B";
  if(s>=60) return "C";
  return "D";
}

function calcScore(detected={}, key={}) {
  let correct=0,wrong=0,empty=0;
  const details={};
  Object.entries(key).forEach(([n,k])=>{
    const g=detected[n];
    if(!g||g==="-"){ empty++; details[n]={status:"empty",key:k,given:"-"}; }
    else if(g===k){ correct++; details[n]={status:"correct",key:k,given:g}; }
    else { wrong++; details[n]={status:"wrong",key:k,given:g}; }
  });
  const total=Object.keys(key).length||1;
  return { correct, wrong, empty, total, score:Math.round((correct/total)*100), details };
}

  const res = await fetch("https://api.anthropic.com/v1/messages",{
    method:"POST",
    headers:{"Content-Type":"application/json"},
    body:JSON.stringify({
      model:"claude-sonnet-4-20250514",

/* ── File Upload ── */
function FileUpload({ onUpload }) {
  const [drag,setDrag]=useState(false);
  const [preview,setPreview]=useState(null);
  const [fname,setFname]=useState("");
  const ref=useRef(null);

  const handle=async(file)=>{
    if(!file) return;
    const ok=["image/jpeg","image/jpg","image/png","application/pdf"];
    if(!ok.includes(file.type)){alert("Gunakan JPG, PNG, atau PDF.");return;}
    const b64=await toB64(file);
    const mt=file.type==="application/pdf"?"application/pdf":"image/jpeg";
    const prev=file.type!=="application/pdf"?URL.createObjectURL(file):null;
    setPreview(prev); setFname(file.name);
    onUpload({b64,mt,preview:prev,fileName:file.name});
  };

  return (
    <div
      onClick={()=>ref.current?.click()}
      onDragOver={e=>{e.preventDefault();setDrag(true);}}
      onDragLeave={()=>setDrag(false)}
      onDrop={e=>{e.preventDefault();setDrag(false);handle(e.dataTransfer.files[0]);}}
      style={{
        borderRadius:20,border:`2px dashed ${drag?"rgba(0,229,160,.6)":"rgba(255,255,255,.12)"}`,
        background: drag?"rgba(0,229,160,.05)":"var(--surface)",
        padding:preview?"12px":"36px 20px",
        textAlign:"center",cursor:"pointer",transition:"all .2s",
        minHeight:preview?0:160,display:"flex",flexDirection:"column",
        alignItems:"center",justifyContent:"center",gap:10,
      }}>
      <input ref={ref} type="file" accept=".jpg,.jpeg,.png,.pdf"
        style={{display:"none"}} onChange={e=>handle(e.target.files[0])}/>
      {preview ? (
        <>
          <img src={preview} style={{maxHeight:240,width:"100%",objectFit:"contain",

/* ── Score Donut ── */
function ScoreDonut({ score, size=140 }) {
  const r=52, circ=2*Math.PI*r;
  const dash=(score/100)*circ;
  const color = score>=90?"#00e5a0":score>=80?"#a3ff5a":score>=70?"#ffb830":score>=60?"#ff8c42":"#ff4d6a";
  return (
    <svg width={size} height={size} viewBox="0 0 120 120" style={{display:"block"}}>
      {/* Track */}
      <circle cx="60" cy="60" r={r} fill="none"
        stroke="rgba(255,255,255,.06)" strokeWidth="10"/>
      {/* Fill */}
      <circle cx="60" cy="60" r={r} fill="none"
        stroke={color} strokeWidth="10"
        strokeLinecap="round"
        strokeDasharray={`${dash} ${circ}`}
        transform="rotate(-90 60 60)"
        style={{
          filter:`drop-shadow(0 0 8px ${color})`,
          animation:"ringFill .9s ease both",
          strokeDashoffset:0,
        }}/>
      {/* Score text */}
      <text x="60" y="56" textAnchor="middle" fill="white"
        style={{fontFamily:"var(--font-mono)",fontSize:22,fontWeight:700}}>
        {score}
      </text>
      <text x="60" y="70" textAnchor="middle" fill="rgba(255,255,255,.35)"
        style={{fontFamily:"var(--font-mono)",fontSize:10}}>
        /100
      </text>
    </svg>
  );
}

/* ── PAGE: SCANNER ── */
function PageScanner({ answerKey, onResult }) {
  const [mode,setMode]=useState("omr");
  const [inputType,setInputType]=useState("camera");
  const [image,setImage]=useState(null);
  const [rubric,setRubric]=useState("Nilai berdasarkan kelengkapan, kedalaman analisis, ketepatan, dan tata bahasa yang digunakan.");
  const [processing,setProcessing]=useState(false);
  const [err,setErr]=useState("");

  const analyze = async()=>{
    if(!image){setErr("Belum ada gambar.");return;}
    if(mode==="omr"&&Object.keys(answerKey).length===0){
      setErr("Kunci jawaban belum diisi. Buka tab Kunci terlebih dahulu.");return;
    }
    setProcessing(true);setErr("");
    try{
      const raw=await callClaude({b64:image.b64,mt:image.mt,answerKey,mode,rubric});
      if(raw.error){setErr("AI gagal membaca: "+raw.raw?.slice(0,100));setProcessing(false);return;}
      if(mode==="omr"){
        const scored=calcScore(raw.detectedAnswers||{},answerKey);
        onResult({...scored,type:"omr",confidence:raw.confidence,imageQuality:raw.imageQuality,
          emptyNumbers:raw.emptyNumbers,doubleAnswers:raw.doubleAnswers});
      } else {
        onResult({
          type:"essay",score:raw.score||0,maxScore:raw.maxScore||100,
          grade:raw.grade,feedback:raw.feedback,criteria:raw.criteria||[],
          strengths:raw.strengths||[],improvements:raw.improvements||[],
          extractedText:raw.extractedText,correct:"-",wrong:"-",empty:"-",
        });
      }
    }catch(e){setErr("Error: "+e.message);}
    setProcessing(false);
  };

  return (
    <div style={{display:"flex",flexDirection:"column",gap:18}}>

      {/* Mode toggle */}
      <div className="anim-fade-up stagger-1">
        <div style={{fontFamily:"var(--font-mono)",fontSize:10,color:"var(--dim)",
          letterSpacing:".1em",marginBottom:8}}>MODE KOREKSI</div>
        <div className="toggle-wrap">
          {[["omr","🔵 Pilihan Ganda (OMR)"],["essay","📝 Esai / Uraian"]].map(([m,l])=>(
            <button key={m} className={`toggle-item ${mode===m?"on":""}`} onClick={()=>setMode(m)}>{l}</button>
          ))}
        </div>
      </div>

      {/* Essay rubric */}
      {mode==="essay"&&(
        <div className="anim-fade-up stagger-2">
          <div style={{fontFamily:"var(--font-mono)",fontSize:10,color:"var(--dim)",
            letterSpacing:".1em",marginBottom:8}}>RUBRIK PENILAIAN</div>
          <textarea className="field" value={rubric} onChange={e=>setRubric(e.target.value)}
            placeholder="Deskripsikan kriteria penilaian..." rows={3}/>
        </div>
      )}

      {/* Input toggle */}
      <div className="anim-fade-up stagger-2">
        <div style={{fontFamily:"var(--font-mono)",fontSize:10,color:"var(--dim)",
          letterSpacing:".1em",marginBottom:8}}>SUMBER GAMBAR</div>
        <div className="toggle-wrap">
          {[["camera","📷 Kamera Real-Time"],["upload","📂 Upload File"]].map(([m,l])=>(
            <button key={m} className={`toggle-item ${inputType===m?"on":""}`}
              onClick={()=>setInputType(m)}>{l}</button>
          ))}
        </div>
      </div>

      {/* Input area */}
      <div className="anim-fade-up stagger-3">
        {inputType==="camera"
          ? <CameraView onCapture={img=>setImage(img)}/>
          : <FileUpload onUpload={img=>setImage(img)}/>
        }
      </div>

      {/* Error */}
      {err&&(
        <div style={{padding:"12px 16px",borderRadius:12,
          background:"rgba(255,77,106,.08)",border:"1px solid rgba(255,77,106,.25)",
          fontFamily:"var(--font-mono)",fontSize:12,color:"var(--danger)"}}>
          ⚠ {err}
        </div>
      )}

      {/* Analyze button */}
      {image&&(
        <div className="anim-fade-up stagger-4">
          <button className="btn-neon" onClick={analyze} disabled={processing}
            style={{width:"100%",padding:"17px",fontSize:16,
              boxShadow:processing?"none":"0 0 40px rgba(0,229,160,.3)"}}>
            {processing?(
              <><span className="spin" style={{marginRight:10,display:"inline-block"}}>⚙</span>
              Menganalisis dengan Claude AI...</>
            ):(
              <>🤖 &nbsp; Analisis Sekarang</>
            )}
          </button>
        </div>
      )}
    </div>
  );
}

              <div style={{fontFamily:"var(--font-mono)",fontSize:10,
                color:"var(--dim)",width:72,flexShrink:0}}>{r}</div>
              <div style={{flex:1,height:22,borderRadius:8,
                background:"rgba(255,255,255,.04)",overflow:"hidden"}}>
                <div style={{
                  height:"100%",borderRadius:8,
                  background:`linear-gradient(90deg,${c},${c}bb)`,
                  width:`${(n/maxN)*100}%`,minWidth:n>0?28:0,
                  boxShadow:`0 0 12px ${c}44`,
                  display:"flex",alignItems:"center",
                  paddingLeft:n>0?8:0,
                  transition:"width 1s ease",
                  fontFamily:"var(--font-mono)",fontSize:11,
                  fontWeight:700,color:"#050810",
                }}>
                  {n>0?n:""}
                </div>
              </div>
            </div>
          ))}
        </div>
      </div>

      {/* Leaderboard */}
      <div className="anim-fade-up stagger-3">
        <div style={{fontFamily:"var(--font-mono)",fontSize:10,color:"var(--dim)",
          letterSpacing:".1em",marginBottom:10}}>RANKING SISWA</div>
        <div style={{display:"flex",flexDirection:"column",gap:6}}>
          {all.map((s,i)=>{
            const c=s.score>=90?"#00e5a0":s.score>=80?"#a3ff5a":s.score>=70?"#ffb830":s.score>=60?"#ff8c42":"#ff4d6a";
            const medal=i===0?"🥇":i===1?"🥈":i===2?"🥉":null;
            return(
              <div key={s.id||i} className="glass"
                style={{padding:"13px 16px",display:"flex",alignItems:"center",
                  gap:12,borderRadius:16,
                  background:i<3?"rgba(255,255,255,.05)":"var(--surface)",
                  borderColor:i===0?"rgba(0,229,160,.2)":"var(--border)"}}>
                <div style={{fontFamily:"var(--font-mono)",fontWeight:700,
                  fontSize:i<3?18:13,width:24,textAlign:"center",
                  color:i<3?c:"var(--dim)"}}>
                  {medal||i+1}
                </div>
                <div style={{flex:1,minWidth:0}}>
                  <div style={{fontFamily:"var(--font-display)",fontWeight:600,
                    fontSize:14,whiteSpace:"nowrap",overflow:"hidden",
                    textOverflow:"ellipsis"}}>{s.name}</div>
                  <div style={{fontFamily:"var(--font-mono)",fontSize:10,
                    color:"var(--dim)",marginTop:2}}>{s.date}</div>
                </div>
                <div style={{textAlign:"right"}}>
                  <div style={{fontFamily:"var(--font-mono)",fontWeight:700,
                    fontSize:22,color:c,
                    textShadow:`0 0 12px ${c}55`}}>{s.score}</div>
                  <div style={{fontFamily:"var(--font-mono)",fontSize:10,
                    color:"var(--dim)",marginTop:1}}>
                    {s.correct!==undefined&&`${s.correct}B `}{s.wrong!==undefined&&`${s.wrong}S`}
                  </div>
                </div>
              </div>
            );
          })}
        </div>
      </div>
    </div>
  );
}

/* ═══════════════════════════════════════════════════════════════
   ROOT APP
═══════════════════════════════════════════════════════════════ */
export default function App() {
  const [tab,setTab]         = useState(0);
  const [result,setResult]   = useState(null);
  const [history,setHistory] = useState([]);
  const [totalQ,setTotalQ]   = useState(20);
  const [answerKey,setAnswerKey] = useState(()=>{
    const k={};
    for(let i=1;i<=20;i++) k[i]="A";
    return k;
  });

  // Sync key to totalQ
  useEffect(()=>{
    setAnswerKey(k=>{
      const nk={};
      for(let i=1;i<=totalQ;i++) nk[i]=k[i]||"A";
      return nk;
    });
  },[totalQ]);

  const handleResult=(r)=>{ setResult(r); setTab(2); };
  const handleSave=(r)=>{ setHistory(h=>[r,...h]); };

  return (
    <>
      <style>{GLOBAL_CSS}</style>
      <BgGrid/>
      <div style={{position:"relative",zIndex:1,minHeight:"100vh"}}>
        <Header active={tab} setActive={setTab}/>
        <main style={{maxWidth:680,margin:"0 auto",padding:"20px 16px 80px"}}>
          {tab===0&&<PageScanner answerKey={answerKey} onResult={handleResult}/>}
          {tab===1&&<PageKunci answerKey={answerKey} setAnswerKey={setAnswerKey}
            totalQ={totalQ} setTotalQ={setTotalQ}/>}
          {tab===2&&<PageHasil result={result} onSave={handleSave}/>}
          {tab===3&&<PageDashboard history={history}/>}
        </main>
      </div>
    </>
  );
}
